Code Example

// Initialize MindfulMeets const mindful = new HoloSphere('MindfulMeets'); // Create a meditation group await mindful.put('oak-grove-circle', 'groups', { id: 'morning-meditation', name: 'Morning Meditation Circle', tradition: 'vipassana', description: 'Daily silent sitting practice', facilitator: 'Sarah Chen', active: true }); // Schedule a gathering await mindful.put('oak-grove-circle', 'gatherings', { id: 'full-moon-sit-nov', group_id: 'morning-meditation', title: 'Full Moon Meditation', date: new Date('2025-11-15T19:00').getTime(), duration: 90, location: 'Riverside Garden', notes: 'Followed by tea and silent reflection', rsvps: [] }); // RSVP to a gathering await mindful.put('oak-grove-circle', 'rsvps', { gathering_id: 'full-moon-sit-nov', participant: 'user-123', timestamp: Date.now() }); // Find all gatherings in an area const gatherings = await mindful.getAll('oak-grove-circle', 'gatherings'); // Federate with StoryCircle to share wisdom teachings await mindful.federate('oak-grove-circle', 'storycircle-wisdom'); // Federate with NeighborHub to access meditation spaces await mindful.federate('oak-grove-circle', 'neighborhub-spaces');
